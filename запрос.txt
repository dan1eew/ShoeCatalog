CREATE TABLE Products (
    Id INT PRIMARY KEY IDENTITY(1,1),
    Name NVARCHAR(200),
    Category NVARCHAR(100),
    Description NVARCHAR(500),
    Manufacturer NVARCHAR(150),
    Supplier NVARCHAR(150),
    Price DECIMAL(10,2),
    Unit NVARCHAR(20),
    Quantity INT,
    Discount DECIMAL(5,2),
    ImagePath NVARCHAR(500)
);
